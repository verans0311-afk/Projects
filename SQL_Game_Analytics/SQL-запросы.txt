Динамика увеличения пользовательской базы:
select date_trunc('month', reg_date) as month_reg, count(*) as count_users
from skygame.users
group by month_reg
order by month_reg


Динамика средней длительности одной игровой сессии:
select  date_trunc('month', start_session) as month, 
	avg(end_session - start_session) as avg_time_session,
	avg(extract(epoch from end_session - start_session))/(60*60) as agg_seconds
from skygame.game_sessions
where end_session - start_session > interval '5 minute'
group by month
order by month

select sum(case when end_session is null then 1.0 else 0.0 end) as count_pr, 
	sum(case when end_session is null then 1.0 else 0.0 end) / count(*) as share_pr
from skygame.game_sessions

select dev_type, sum(case when end_session is null then 1.0 else 0.0 end) / count(*) as share_pr
from skygame.users us
join skygame.game_sessions gs
	on us.id_user = gs.id_user
group by dev_type

select sum(case when dev_type = 'ios' then 1.0 else 0.0 end) / count(*) as ios_pr, 
	sum(case when dev_type = 'android' then 1.0 else 0.0 end) / count(*) as android_pr
from skygame.users us
join skygame.game_sessions gs
	on us.id_user = gs.id_user
where end_session is null

Динамика суммарных клиентских выплат в разрезе типов продуктов:
select date_trunc('month', dtime_pay) as month, type, sum(cnt_buy * price) as revenue
from skygame.monetary m
join skygame.item_list il
	on m.id_item_buy = il.id_item
join skygame.log_prices lp
	on m.id_item_buy = lp.id_item
	and dtime_pay::date >= valid_from and dtime_pay::date < coalesce(valid_to, to_date('3000-01-01', 'YYYY-MM-DD'))
group by month, type
order by type, month

Анализ средних платежей по когортам:
select *, extract('day' from (select max(dtime_pay) from skygame.monetary) - month) / 30 as interval
from
	(select date_trunc('month', reg_date) as month, 
		sum(cnt_buy * price) as revenue, 
		count(distinct u.id_user) as count_users, 
		sum(cnt_buy * price) / count(distinct u.id_user) as avg_revenue
	from skygame.monetary m
	join skygame.users u
		on m.id_user = u.id_user
	join skygame.log_prices lp
		on m.id_item_buy = lp.id_item
		and dtime_pay::date >= valid_from and dtime_pay::date < coalesce(valid_to, to_date('3000-01-01', 'YYYY-MM-DD'))
	where reg_date < (select max(dtime_pay) from skygame.monetary) - interval '1 month'
	group by month
	order by month) t

Расчет k-фактора:
with k_f as
	(select sum(ref_reg) / count(distinct u.id_user) as k_factor
	from skygame.users u
	left join skygame.referral r
		on u.id_user = r.id_user),
a_c as
	(select count(*) / count(distinct date_trunc('month', reg_date)) as avg_cohort
	from skygame.users)

select (select * from k_f) as k_factor, (select * from a_c) * (select * from k_f) as predict_amount_users


Анализ альтернативной стратегии привлечения клиентов:
select case when date_trunc('month', reg_date) = '2022-11-01' or date_trunc('month', reg_date) = '2022-12-01' then '11-12.2022' else 'others' end as segment,
        avg(end_session - start_session) as avg_time_session
from skygame.users u
left join skygame.game_sessions gs
	on u.id_user = gs.id_user
where end_session - start_session > interval '5 minute'
group by segment


Анализ лояльности:
with crit_invite as
	(select id_user, count(*) as amount_invite, sum(ref_reg) as amount_reg
	from skygame.referral
	group by id_user
	having count(*) >= 3 and sum(ref_reg) >= 1)
select date_trunc('month', start_session) as month, count(distinct id_user) as count_users
from skygame.game_sessions
where id_user in (select id_user from crit_invite)
group by month
order by month

with crit_1000 as
	(select id_user, sum(cnt_buy * price) as revenue
	from skygame.monetary m
	join skygame.log_prices lp
		on m.id_item_buy = lp.id_item
	where dtime_pay::date >= valid_from and dtime_pay::date < coalesce(valid_to, to_date('3000-01-01', 'YYYY-MM-DD'))
	group by m.id_user
	having sum(cnt_buy * price) > 1000)
select date_trunc('month', start_session) as month, count(distinct id_user) as count_users
from skygame.game_sessions
where id_user in (select id_user from crit_1000)
group by month
order by month

select date_trunc('month', start_session) as month, count(distinct id_user) as count_users
from skygame.game_sessions
where id_user in (select id_user from crit_invite) and id_user in (select id_user from crit_1000)
group by month
order by month

select date_trunc('month', start_session) as month, count(distinct id_user) as count_users
from skygame.game_sessions
where id_user in (select id_user from crit_invite) or id_user in (select id_user from crit_1000)
group by month
order by month

with ltr as
	(select id_user, sum(cnt_buy * price) as revenue
	from skygame.monetary m
	join skygame.log_prices lp
		on m.id_item_buy = lp.id_item
	where dtime_pay::date >= valid_from and dtime_pay::date < coalesce(valid_to, to_date('3000-01-01', 'YYYY-MM-DD'))
	group by m.id_user),
lt_month as
	(select gs.id_user, ceil(extract('days' from max(start_session) - min(reg_date)) / 30) as lt_month
	from skygame.game_sessions gs
	join skygame.users u
		on gs.id_user = u.id_user
	group by gs.id_user)
select ltr.id_user, revenue / lt_month as ltr_month, lt_month
from ltr
join lt_month
	on ltr.id_user = lt_month.id_user
order by ltr_month desc
limit 100
